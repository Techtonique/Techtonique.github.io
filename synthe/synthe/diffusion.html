 132    def _init_noise_schedule(self) -> None:
 133        """Initialize forward diffusion noise schedule with numerical stability"""
 134        if self.schedule == "linear":
 135            self.betas = np.linspace(
 136                self.beta_start, self.beta_end, self.timesteps
 137            )
 138        elif self.schedule == "cosine":
 139            s = 0.008
 140            steps = np.arange(self.timesteps + 1, dtype=np.float64)
 141            alphas_bar = (
 142                np.cos(((steps / self.timesteps) + s) / (1 + s) * np.pi * 0.5)
 143                ** 2
 144            )
 145            alphas_bar = alphas_bar / alphas_bar[0]
 146            self.betas = np.clip(
 147                1 - (alphas_bar[1:] / alphas_bar[:-1]), 0, 0.999
 148            )
 149
 150        self.alphas = 1.0 - self.betas
 151        # Numerical stability: clamp away from 0
 152        self.alphas_cumprod = np.clip(np.cumprod(self.alphas), 1e-8, 1.0)
 153        self.sqrt_alphas_cumprod = np.sqrt(self.alphas_cumprod)
 154        self.sqrt_one_minus_alphas_cumprod = np.sqrt(1.0 - self.alphas_cumprod)
 155
 156        # Compute posterior variance (beta_tilde) for true posterior
 157        alphas_cumprod_prev = np.concatenate([[1.0], self.alphas_cumprod[:-1]])
 158        self.posterior_variance = (
 159            self.betas
 160            * (1.0 - alphas_cumprod_prev)
 161            / (1.0 - self.alphas_cumprod)
 162        )
 163
 164        # Compute posterior mean coefficients
 165        self.posterior_mean_coef1 = (
 166            self.betas
 167            * np.sqrt(alphas_cumprod_prev)
 168            / (1.0 - self.alphas_cumprod)
 169        )
 170        self.posterior_mean_coef2 = (
 171            (1.0 - alphas_cumprod_prev)
 172            * np.sqrt(self.alphas)
 173            / (1.0 - self.alphas_cumprod)
 174        )

 195    def forward_diffusion(
 196        self, x0: np.ndarray, t: np.ndarray, noise: Optional[np.ndarray] = None
 197    ) -> Tuple[np.ndarray, np.ndarray]:
 198        """
 199        Forward diffusion: q(x_t | x_0) = N(sqrt(alpha_bar_t) * x_0, (1 - alpha_bar_t) * I)
 200
 201        Parameters
 202        ----------
 203        x0 : ndarray of shape (n_samples, n_features)
 204            Original data samples
 205        t : ndarray of shape (n_samples,)
 206            Timesteps (0 to timesteps-1)
 207        noise : ndarray, optional
 208            Noise to add (generated if None)
 209
 210        Returns
 211        -------
 212        xt : ndarray
 213            Noised data
 214        noise : ndarray
 215            Noise that was added
 216        """
 217        x0 = self._validate_data(x0)
 218        self._validate_timestep(t)
 219
 220        if noise is None:
 221            noise = self._rng.randn(*x0.shape)
 222
 223        # Vectorized computation for batch processing
 224        sqrt_alpha = self.sqrt_alphas_cumprod[t][:, np.newaxis]
 225        sqrt_one_minus_alpha = self.sqrt_one_minus_alphas_cumprod[t][
 226            :, np.newaxis
 227        ]
 228
 229        xt = sqrt_alpha * x0 + sqrt_one_minus_alpha * noise
 230        return xt, noise

 232    def _sample_true_posterior(
 233        self, x0: np.ndarray, xt: np.ndarray, t: int, n_samples: int = 1
 234    ) -> np.ndarray:
 235        """
 236        Sample from true posterior q(x_{t-1} | x_t, x_0)
 237
 238        This is the ground truth denoising distribution that we want to match.
 239
 240        Parameters
 241        ----------
 242        x0 : ndarray of shape (batch_size, n_features)
 243            Clean data
 244        xt : ndarray of shape (batch_size, n_features)
 245            Noisy data at timestep t
 246        t : int
 247            Current timestep
 248        n_samples : int
 249            Number of samples to draw per data point
 250
 251        Returns
 252        -------
 253        samples : ndarray of shape (batch_size * n_samples, n_features)
 254            Samples from true posterior
 255        """
 256        if t == 0:
 257            # At t=0, posterior is deterministic: x_{-1} doesn't exist, return x_0
 258            return np.repeat(x0, n_samples, axis=0)
 259
 260        batch_size = x0.shape[0]
 261
 262        # Posterior mean: μ̃_t = coef1 * x_0 + coef2 * x_t
 263        coef1 = self.posterior_mean_coef1[t]
 264        coef2 = self.posterior_mean_coef2[t]
 265        posterior_mean = coef1 * x0 + coef2 * xt
 266
 267        # Posterior variance: σ̃_t²
 268        posterior_var = self.posterior_variance[t]
 269
 270        # Sample from N(μ̃_t, σ̃_t² I)
 271        samples = []
 272        for _ in range(n_samples):
 273            noise = self._rng.randn(*x0.shape) * np.sqrt(posterior_var)
 274            sample = posterior_mean + noise
 275            samples.append(sample)
 276
 277        return np.vstack(samples)

 279    def _sample_learned_transition(
 280        self, xt: np.ndarray, t: int, n_samples: int = 1
 281    ) -> np.ndarray:
 282        """
 283        Sample from learned transition p_θ(x_{t-1} | x_t)
 284
 285        Parameters
 286        ----------
 287        xt : ndarray of shape (batch_size, n_features)
 288            Noisy data at timestep t
 289        t : int
 290            Current timestep
 291        n_samples : int
 292            Number of samples to draw per data point
 293
 294        Returns
 295        -------
 296        samples : ndarray of shape (batch_size * n_samples, n_features)
 297            Samples from learned transition
 298        """
 299        batch_size = xt.shape[0]
 300        t_array = np.full(batch_size, t)
 301
 302        # Predict noise using learned model
 303        features = self._create_features(xt, t_array)
 304        pred_noise = self.model_.predict(features).reshape(xt.shape)
 305
 306        # Compute predicted mean
 307        alpha = self.alphas[t]
 308        alpha_bar = self.alphas_cumprod[t]
 309        beta = self.betas[t]
 310
 311        coef1 = 1.0 / np.sqrt(alpha)
 312        coef2 = beta / np.sqrt(1.0 - alpha_bar)
 313        predicted_mean = coef1 * (xt - coef2 * pred_noise)
 314
 315        # Get variance
 316        if self.variance_type == "fixed_small":
 317            variance = self.posterior_variance[t]
 318        elif self.variance_type == "fixed_large":
 319            variance = beta
 320        else:  # learned - for now use fixed_small
 321            variance = self.posterior_variance[t]
 322
 323        # Sample from N(μ_θ, σ² I)
 324        if t == 0:
 325            # Deterministic at final step
 326            return np.repeat(predicted_mean, n_samples, axis=0)
 327
 328        samples = []
 329        for _ in range(n_samples):
 330            noise = self._rng.randn(*xt.shape) * np.sqrt(variance)
 331            sample = predicted_mean + noise
 332            samples.append(sample)
 333
 334        return np.vstack(samples)

 336    def _compute_kernel(
 337        self, X: np.ndarray, Y: np.ndarray, gamma: Optional[float] = None
 338    ) -> np.ndarray:
 339        """
 340        Compute kernel matrix K(X, Y) using specified kernel
 341
 342        Parameters
 343        ----------
 344        X : ndarray of shape (n, d)
 345        Y : ndarray of shape (m, d)
 346        gamma : float, optional
 347            Kernel bandwidth
 348
 349        Returns
 350        -------
 351        K : ndarray of shape (n, m)
 352            Kernel matrix
 353        """
 354        if self.mmd_kernel == "linear":
 355            return X @ Y.T
 356
 357        # Compute squared distances
 358        try:
 359            from scipy.spatial.distance import cdist
 360
 361            sq_dist = cdist(X, Y, metric="sqeuclidean")
 362        except ImportError:
 363            sq_dist = np.sum((X[:, None, :] - Y[None, :, :]) ** 2, axis=2)
 364
 365        if gamma is None:
 366            # Use median heuristic
 367            if len(X) > 100:
 368                sample_idx = self._rng.choice(len(X), 100, replace=False)
 369                sq_dist_sample = sq_dist[sample_idx][:, :100]
 370            else:
 371                sq_dist_sample = sq_dist
 372            gamma = 1.0 / np.median(sq_dist_sample[sq_dist_sample > 0])
 373
 374        if self.mmd_kernel == "rbf":
 375            return np.exp(-gamma * sq_dist)
 376        elif self.mmd_kernel == "imq":
 377            # Inverse multiquadric: (||x-y||² + c²)^(-β)
 378            c = 1.0
 379            beta = 0.5
 380            return (sq_dist + c**2) ** (-beta)
 381        else:
 382            raise ValueError(f"Unknown kernel: {self.mmd_kernel}")

 384    def _compute_mmd_unbiased(
 385        self, X: np.ndarray, Y: np.ndarray, gamma: Optional[float] = None
 386    ) -> float:
 387        """
 388        Compute unbiased MMD² estimator
 389
 390        MMD²(P, Q) = E[k(u,u')] + E[k(v,v')] - 2E[k(u,v)]
 391
 392        Uses unbiased estimator that excludes diagonal terms.
 393        """
 394        n, m = len(X), len(Y)
 395
 396        # Compute kernel matrices
 397        K_XX = self._compute_kernel(X, X, gamma)
 398        K_YY = self._compute_kernel(Y, Y, gamma)
 399        K_XY = self._compute_kernel(X, Y, gamma)
 400
 401        # Unbiased estimator: exclude diagonal
 402        if n > 1:
 403            K_XX_unbiased = (K_XX.sum() - np.trace(K_XX)) / (n * (n - 1))
 404        else:
 405            K_XX_unbiased = 0
 406
 407        if m > 1:
 408            K_YY_unbiased = (K_YY.sum() - np.trace(K_YY)) / (m * (m - 1))
 409        else:
 410            K_YY_unbiased = 0
 411
 412        K_XY_mean = K_XY.mean()
 413
 414        mmd_sq = K_XX_unbiased + K_YY_unbiased - 2 * K_XY_mean
 415
 416        return max(0.0, mmd_sq)  # Ensure non-negative

 418    def _create_features(self, x: np.ndarray, t: np.ndarray) -> np.ndarray:
 419        """
 420        Create features for the regressor with multi-frequency temporal encoding
 421        """
 422        # Normalize timestep
 423        t_norm = t / self.timesteps
 424
 425        # Multi-frequency positional encoding
 426        frequencies = [1.0, 2.0, 4.0]
 427        t_encoded_parts = []
 428
 429        for freq in frequencies:
 430            t_encoded_parts.extend(
 431                [
 432                    np.sin(2 * np.pi * freq * t_norm),
 433                    np.cos(2 * np.pi * freq * t_norm),
 434                ]
 435            )
 436
 437        t_encoded_parts.append(t_norm)
 438        t_encoded = np.column_stack(t_encoded_parts)
 439
 440        # Concatenate spatial and temporal features
 441        features = np.hstack([x, t_encoded])
 442        return features

 444    def fit(
 445        self,
 446        X: np.ndarray,
 447        y: Optional[np.ndarray] = None,
 448        n_steps: int = 1000,
 449    ) -> "DiffusionModel":
 450        """
 451        Train the reverse diffusion model using specified objective
 452
 453        Parameters
 454        ----------
 455        X : ndarray of shape (n_samples, n_features)
 456            Training data (will be normalized internally)
 457        y : ignored
 458            Not used, present for API consistency
 459        n_steps : int, default=1000
 460            Number of training iterations
 461
 462        Returns
 463        -------
 464        self : DiffusionModel
 465            Fitted estimator
 466        """
 467        X = self._validate_data(X)
 468        n_samples, n_features = X.shape
 469
 470        # Store normalization params
 471        self.X_mean_ = X.mean(axis=0)
 472        self.X_std_ = X.std(axis=0) + 1e-8
 473        X_norm = (X - self.X_mean_) / self.X_std_
 474
 475        # Optional PCA
 476        if self.use_pca and n_features > 100:
 477            self.pca_ = PCA(
 478                n_components=min(self.pca_components, n_features),
 479                random_state=self.random_state,
 480            )
 481            X_norm = self.pca_.fit_transform(X_norm)
 482            n_features = X_norm.shape[1]
 483        else:
 484            self.pca_ = None
 485
 486        self.n_features_ = n_features
 487
 488        # Initialize model
 489        if self.model is None:
 490            self.model_ = Ridge(alpha=1.0, random_state=self.random_state)
 491        else:
 492            self.model_ = clone(self.model)
 493
 494        # Train based on objective
 495        if self.training_objective == "noise":
 496            self._fit_noise_prediction(X_norm, n_samples, n_steps)
 497        elif self.training_objective == "mmd":
 498            self._fit_mmd_matching(X_norm, n_samples, n_steps)
 499        elif self.training_objective == "hybrid":
 500            self._fit_hybrid(X_norm, n_samples, n_steps)
 501
 502        self.is_fitted_ = True
 503        return self

 505    def _fit_noise_prediction(
 506        self, X_norm: np.ndarray, n_samples: int, n_steps: int
 507    ) -> None:
 508        """Traditional DDPM training: predict noise with MSE loss"""
 509        n_batches = max(1, n_steps // self.batch_size)
 510
 511        X_train_batches: List[np.ndarray] = []
 512        y_train_batches: List[np.ndarray] = []
 513
 514        for batch_idx in range(n_batches):
 515            indices = self._rng.randint(n_samples, size=self.batch_size)
 516            t_batch = self._rng.randint(self.timesteps, size=self.batch_size)
 517
 518            x0_batch = X_norm[indices]
 519            noise_batch = self._rng.randn(self.batch_size, self.n_features_)
 520
 521            xt_batch, _ = self.forward_diffusion(x0_batch, t_batch, noise_batch)
 522            features = self._create_features(xt_batch, t_batch)
 523
 524            X_train_batches.append(features)
 525            y_train_batches.append(noise_batch)
 526
 527        X_train = np.vstack(X_train_batches)
 528        y_train = np.vstack(y_train_batches)
 529
 530        self.model_.fit(X_train, y_train)

 532    def _fit_mmd_matching(
 533        self, X_norm: np.ndarray, n_samples: int, n_steps: int
 534    ) -> None:
 535        """
 536        MMD-based training: directly minimize distribution mismatch
 537
 538        This implements the algorithm from the theoretical document:
 539        1. Sample (x_0, t, x_t)
 540        2. Draw samples from true posterior q(x_{t-1}|x_t,x_0)
 541        3. Draw samples from learned transition p_θ(x_{t-1}|x_t)
 542        4. Compute MMD² between these sample sets
 543        5. Update θ to minimize MMD
 544        """
 545        print(
 546            f"Training with MMD objective (kernel={self.mmd_kernel}, samples={self.mmd_samples_per_step})"
 547        )
 548
 549        # Determine gamma for kernel
 550        if self.mmd_bandwidth == "auto":
 551            gamma = None  # Will be computed adaptively
 552        else:
 553            gamma = float(self.mmd_bandwidth)
 554
 555        # For sklearn models without iterative updates, we need to:
 556        # 1. Generate training data with MMD-weighted importance
 557        # 2. Fit the model on this data
 558
 559        # We'll use an iterative refinement approach:
 560        # Start with noise prediction, then refine with MMD
 561        print("Phase 1: Initial noise prediction training...")
 562        self._fit_noise_prediction(X_norm, n_samples, n_steps // 2)
 563
 564        print("Phase 2: MMD refinement...")
 565        # Generate MMD-weighted training data
 566        X_train_batches: List[np.ndarray] = []
 567        y_train_batches: List[np.ndarray] = []
 568        weights_batches: List[np.ndarray] = []
 569
 570        n_batches = max(1, (n_steps // 2) // self.batch_size)
 571
 572        for batch_idx in range(n_batches):
 573            if batch_idx % 10 == 0:
 574                print(f"  Batch {batch_idx}/{n_batches}")
 575
 576            indices = self._rng.randint(n_samples, size=self.batch_size)
 577            t_batch = self._rng.randint(1, self.timesteps, size=self.batch_size)
 578
 579            x0_batch = X_norm[indices]
 580            noise_batch = self._rng.randn(self.batch_size, self.n_features_)
 581            xt_batch, _ = self.forward_diffusion(x0_batch, t_batch, noise_batch)
 582
 583            # For each timestep, compute MMD between true and learned transitions
 584            batch_weights = []
 585            for i in range(self.batch_size):
 586                t = t_batch[i]
 587                x0_i = x0_batch[i: i + 1]
 588                xt_i = xt_batch[i: i + 1]
 589
 590                # Sample from both distributions
 591                true_samples = self._sample_true_posterior(
 592                    x0_i, xt_i, t, n_samples=self.mmd_samples_per_step
 593                )
 594                learned_samples = self._sample_learned_transition(
 595                    xt_i, t, n_samples=self.mmd_samples_per_step
 596                )
 597
 598                # Compute MMD
 599                mmd = self._compute_mmd_unbiased(
 600                    true_samples, learned_samples, gamma
 601                )
 602
 603                # Weight by MMD (higher mismatch = higher weight)
 604                weight = 1.0 + mmd
 605                batch_weights.append(weight)
 606
 607            features = self._create_features(xt_batch, t_batch)
 608            X_train_batches.append(features)
 609            y_train_batches.append(noise_batch)
 610            weights_batches.append(np.array(batch_weights))
 611
 612        # Fit with weighted samples
 613        X_train = np.vstack(X_train_batches)
 614        y_train = np.vstack(y_train_batches)
 615        weights = np.concatenate(weights_batches)
 616
 617        # Normalize weights
 618        weights = weights / weights.sum() * len(weights)
 619
 620        print("  Final weighted fit...")
 621        if (
 622            hasattr(self.model_, "fit")
 623            and "sample_weight" in self.model_.fit.__code__.co_varnames
 624        ):

 654    def sample(
 655        self,
 656        n_samples: int = 1,
 657        return_trajectory: bool = False,
 658        ddim: bool = False,
 659        ddim_steps: int = 50,
 660    ) -> np.ndarray:
 661        """
 662        Generate samples via reverse diffusion
 663
 664        Parameters
 665        ----------
 666        n_samples : int, default=1
 667            Number of samples to generate
 668        return_trajectory : bool, default=False
 669            If True, return all intermediate denoising steps
 670        ddim : bool, default=False
 671            Use DDIM for faster deterministic sampling
 672        ddim_steps : int, default=50
 673            Number of DDIM steps (only used if ddim=True)
 674
 675        Returns
 676        -------
 677        samples : ndarray
 678            Generated samples (denormalized to original scale)
 679        """
 680        check_is_fitted(self, ["model_", "X_mean_", "X_std_", "n_features_"])
 681
 682        if ddim:
 683            return self._sample_ddim(n_samples, ddim_steps, return_trajectory)
 684
 685        # Standard DDPM sampling
 686        x = self._rng.randn(n_samples, self.n_features_)
 687        trajectory = [x.copy()] if return_trajectory else None
 688
 689        for t in reversed(range(self.timesteps)):
 690            t_array = np.full(n_samples, t)
 691            features = self._create_features(x, t_array)
 692
 693            pred_noise = self.model_.predict(features).reshape(x.shape)
 694
 695            alpha = self.alphas[t]
 696            alpha_bar = self.alphas_cumprod[t]
 697            beta = self.betas[t]
 698
 699            coef1 = 1.0 / np.sqrt(alpha)
 700            coef2 = beta / np.sqrt(1.0 - alpha_bar)
 701            mean = coef1 * (x - coef2 * pred_noise)
 702
 703            if t > 0:
 704                if self.variance_type == "fixed_small":
 705                    variance = self.posterior_variance[t]
 706                else:
 707                    variance = beta
 708                noise = np.sqrt(variance) * self._rng.randn(*x.shape)
 709                x = mean + noise
 710            else:
 711                x = mean
 712
 713            if return_trajectory:
 714                trajectory.append(x.copy())
 715
 716        return self._postprocess_samples(x, trajectory, return_trajectory)

 718    def _sample_ddim(
 719        self, n_samples: int, steps: int, return_trajectory: bool
 720    ) -> np.ndarray:
 721        """DDIM sampling for faster generation"""
 722        skip = max(1, self.timesteps // steps)
 723        timesteps = np.arange(0, self.timesteps, skip)[::-1]
 724
 725        x = self._rng.randn(n_samples, self.n_features_)
 726        trajectory = [x.copy()] if return_trajectory else None
 727
 728        for i, t in enumerate(timesteps):
 729            t_array = np.full(n_samples, t)
 730            features = self._create_features(x, t_array)
 731            pred_noise = self.model_.predict(features).reshape(x.shape)
 732
 733            alpha_bar = self.alphas_cumprod[t]
 734            alpha_bar_prev = (
 735                self.alphas_cumprod[timesteps[i + 1]]
 736                if i < len(timesteps) - 1
 737                else 1.0
 738            )
 739
 740            pred_x0 = (x - np.sqrt(1 - alpha_bar) * pred_noise) / np.sqrt(
 741                alpha_bar
 742            )
 743            dir_xt = np.sqrt(1 - alpha_bar_prev) * pred_noise
 744            x = np.sqrt(alpha_bar_prev) * pred_x0 + dir_xt
 745
 746            if return_trajectory:
 747                trajectory.append(x.copy())
 748
 749        return self._postprocess_samples(x, trajectory, return_trajectory)

 751    def _postprocess_samples(
 752        self,
 753        x: np.ndarray,
 754        trajectory: Optional[List[np.ndarray]],
 755        return_trajectory: bool,
 756    ) -> np.ndarray:
 757        """Apply inverse transforms to generated samples"""
 758        if return_trajectory and trajectory is not None:
 759            processed_trajectory = []
 760            for step in trajectory:
 761                if self.pca_ is not None:
 762                    step = self.pca_.inverse_transform(step)
 763                step = step * self.X_std_ + self.X_mean_
 764                processed_trajectory.append(step)
 765            return np.array(processed_trajectory)
 766
 767        if self.pca_ is not None:
 768            x = self.pca_.inverse_transform(x)
 769
 770        x = x * self.X_std_ + self.X_mean_
 771        return x

 821    def _compute_mmd(
 822        self, X: np.ndarray, Y: np.ndarray, gamma: Union[str, float] = "auto"
 823    ) -> float:
 824        """Compute MMD with adaptive bandwidth (for evaluation)"""
 825        max_samples = 1000
 826        if len(X) > max_samples:
 827            indices = self._rng.choice(len(X), max_samples, replace=False)
 828            X = X[indices]
 829        if len(Y) > max_samples:
 830            indices = self._rng.choice(len(Y), max_samples, replace=False)
 831            Y = Y[indices]
 832
 833        if gamma == "auto":
 834            n_bandwidth = min(100, len(X), len(Y))
 835            X_bw = X[:n_bandwidth]
 836            Y_bw = Y[:n_bandwidth]
 837            combined = np.vstack([X_bw, Y_bw])
 838
 839            from scipy.spatial.distance import pdist
 840
 841            squared_dists = pdist(combined, metric="sqeuclidean")
 842            gamma = 1.0 / np.median(squared_dists)
 843
 844        try:
 845            from scipy.spatial.distance import cdist
 846
 847            XX = cdist(X, X, metric="sqeuclidean")
 848            YY = cdist(Y, Y, metric="sqeuclidean")
 849            XY = cdist(X, Y, metric="sqeuclidean")
 850
 851            K_XX = np.exp(-gamma * XX).mean()
 852            K_YY = np.exp(-gamma * YY).mean()
 853            K_XY = np.exp(-gamma * XY).mean()
 854        except ImportError:
 855            chunk_size = 100
 856
 857            def chunked_kernel_mean(A, B):
 858                total = 0.0
 859                count = 0
 860                for i in range(0, len(A), chunk_size):
 861                    A_chunk = A[i: i + chunk_size]
 862                    for j in range(0, len(B), chunk_size):
 863                        B_chunk = B[j: j + chunk_size]
 864                        sq_dist = np.sum(
 865                            (A_chunk[:, None, :] - B_chunk[None, :, :]) ** 2,
 866                            axis=2,
 867                        )
 868                        total += np.exp(-gamma * sq_dist).sum()
 869                        count += len(A_chunk) * len(B_chunk)
 870                return total / count
 871
 872            K_XX = chunked_kernel_mean(X, X)
 873            K_YY = chunked_kernel_mean(Y, Y)
 874            K_XY = chunked_kernel_mean(X, Y)
 875
 876        mmd = K_XX + K_YY - 2 * K_XY
 877        return max(0, mmd)

 879    def _compute_energy_distance(self, X: np.ndarray, Y: np.ndarray) -> float:
 880        """Compute energy distance"""
 881        try:
 882            from scipy.spatial.distance import pdist, cdist
 883
 884            max_samples = 500
 885            if len(X) > max_samples:
 886                indices = self._rng.choice(len(X), max_samples, replace=False)
 887                X = X[indices]
 888            if len(Y) > max_samples:
 889                indices = self._rng.choice(len(Y), max_samples, replace=False)
 890                Y = Y[indices]
 891
 892            xy_dist = cdist(X, Y, metric="euclidean").mean()
 893            xx_dist = pdist(X, metric="euclidean").mean() if len(X) > 1 else 0
 894            yy_dist = pdist(Y, metric="euclidean").mean() if len(Y) > 1 else 0
 895
 896            energy = 2 * xy_dist - xx_dist - yy_dist
 897            return max(0, energy)
 898        except ImportError:
 899            max_samples = 500
 900            if len(X) > max_samples:
 901                indices = self._rng.choice(len(X), max_samples, replace=False)
 902                X = X[indices]
 903            if len(Y) > max_samples:
 904                indices = self._rng.choice(len(Y), max_samples, replace=False)
 905                Y = Y[indices]
 906
 907            def chunked_pairwise_dist(A, B, chunk_size=100):
 908                total = 0.0
 909                count = 0
 910                for i in range(0, len(A), chunk_size):
 911                    A_chunk = A[i: i + chunk_size]
 912                    for j in range(0, len(B), chunk_size):
 913                        B_chunk = B[j: j + chunk_size]
 914                        dist = np.sqrt(
 915                            np.sum(
 916                                (A_chunk[:, None, :] - B_chunk[None, :, :])
 917                                ** 2,
 918                                axis=2,
 919                            )
 920                        )
 921                        total += dist.sum()
 922                        count += len(A_chunk) * len(B_chunk)
 923                return total / count
 924
 925            xy_dist = chunked_pairwise_dist(X, Y)
 926
 927            n, m = len(X), len(Y)
 928            if n > 1:
 929                xx_dist = chunked_pairwise_dist(X, X)
 930                xx_dist = (xx_dist * n * n) / (n * (n - 1))
 931            else:
 932                xx_dist = 0
 933
 934            if m > 1:
 935                yy_dist = chunked_pairwise_dist(Y, Y)
 936                yy_dist = (yy_dist * m * m) / (m * (m - 1))
 937            else:
 938                yy_dist = 0
 939
 940            energy = 2 * xy_dist - xx_dist - yy_dist
 941            return max(0, energy)

 133    def _init_noise_schedule(self) -> None:
 134        """Initialize forward diffusion noise schedule with numerical stability"""
 135        if self.schedule == "linear":
 136            self.betas = np.linspace(
 137                self.beta_start, self.beta_end, self.timesteps
 138            )
 139        elif self.schedule == "cosine":
 140            s = 0.008
 141            steps = np.arange(self.timesteps + 1, dtype=np.float64)
 142            alphas_bar = (
 143                np.cos(((steps / self.timesteps) + s) / (1 + s) * np.pi * 0.5)
 144                ** 2
 145            )
 146            alphas_bar = alphas_bar / alphas_bar[0]
 147            self.betas = np.clip(
 148                1 - (alphas_bar[1:] / alphas_bar[:-1]), 0, 0.999
 149            )
 150
 151        self.alphas = 1.0 - self.betas
 152        # Numerical stability: clamp away from 0
 153        self.alphas_cumprod = np.clip(np.cumprod(self.alphas), 1e-8, 1.0)
 154        self.sqrt_alphas_cumprod = np.sqrt(self.alphas_cumprod)
 155        self.sqrt_one_minus_alphas_cumprod = np.sqrt(1.0 - self.alphas_cumprod)
 156
 157        # Compute posterior variance (beta_tilde) for true posterior
 158        alphas_cumprod_prev = np.concatenate([[1.0], self.alphas_cumprod[:-1]])
 159        self.posterior_variance = (
 160            self.betas
 161            * (1.0 - alphas_cumprod_prev)
 162            / (1.0 - self.alphas_cumprod)
 163        )
 164
 165        # Compute posterior mean coefficients
 166        self.posterior_mean_coef1 = (
 167            self.betas
 168            * np.sqrt(alphas_cumprod_prev)
 169            / (1.0 - self.alphas_cumprod)
 170        )
 171        self.posterior_mean_coef2 = (
 172            (1.0 - alphas_cumprod_prev)
 173            * np.sqrt(self.alphas)
 174            / (1.0 - self.alphas_cumprod)
 175        )

 196    def forward_diffusion(
 197        self, x0: np.ndarray, t: np.ndarray, noise: Optional[np.ndarray] = None
 198    ) -> Tuple[np.ndarray, np.ndarray]:
 199        """
 200        Forward diffusion: q(x_t | x_0) = N(sqrt(alpha_bar_t) * x_0, (1 - alpha_bar_t) * I)
 201
 202        Parameters
 203        ----------
 204        x0 : ndarray of shape (n_samples, n_features)
 205            Original data samples
 206        t : ndarray of shape (n_samples,)
 207            Timesteps (0 to timesteps-1)
 208        noise : ndarray, optional
 209            Noise to add (generated if None)
 210
 211        Returns
 212        -------
 213        xt : ndarray
 214            Noised data
 215        noise : ndarray
 216            Noise that was added
 217        """
 218        x0 = self._validate_data(x0)
 219        self._validate_timestep(t)
 220
 221        if noise is None:
 222            noise = self._rng.randn(*x0.shape)
 223
 224        # Vectorized computation for batch processing
 225        sqrt_alpha = self.sqrt_alphas_cumprod[t][:, np.newaxis]
 226        sqrt_one_minus_alpha = self.sqrt_one_minus_alphas_cumprod[t][
 227            :, np.newaxis
 228        ]
 229
 230        xt = sqrt_alpha * x0 + sqrt_one_minus_alpha * noise
 231        return xt, noise

 233    def _sample_true_posterior(
 234        self, x0: np.ndarray, xt: np.ndarray, t: int, n_samples: int = 1
 235    ) -> np.ndarray:
 236        """
 237        Sample from true posterior q(x_{t-1} | x_t, x_0)
 238
 239        This is the ground truth denoising distribution that we want to match.
 240
 241        Parameters
 242        ----------
 243        x0 : ndarray of shape (batch_size, n_features)
 244            Clean data
 245        xt : ndarray of shape (batch_size, n_features)
 246            Noisy data at timestep t
 247        t : int
 248            Current timestep
 249        n_samples : int
 250            Number of samples to draw per data point
 251
 252        Returns
 253        -------
 254        samples : ndarray of shape (batch_size * n_samples, n_features)
 255            Samples from true posterior
 256        """
 257        if t == 0:
 258            # At t=0, posterior is deterministic: x_{-1} doesn't exist, return x_0
 259            return np.repeat(x0, n_samples, axis=0)
 260
 261        batch_size = x0.shape[0]
 262
 263        # Posterior mean: μ̃_t = coef1 * x_0 + coef2 * x_t
 264        coef1 = self.posterior_mean_coef1[t]
 265        coef2 = self.posterior_mean_coef2[t]
 266        posterior_mean = coef1 * x0 + coef2 * xt
 267
 268        # Posterior variance: σ̃_t²
 269        posterior_var = self.posterior_variance[t]
 270
 271        # Sample from N(μ̃_t, σ̃_t² I)
 272        samples = []
 273        for _ in range(n_samples):
 274            noise = self._rng.randn(*x0.shape) * np.sqrt(posterior_var)
 275            sample = posterior_mean + noise
 276            samples.append(sample)
 277
 278        return np.vstack(samples)

 280    def _sample_learned_transition(
 281        self, xt: np.ndarray, t: int, n_samples: int = 1
 282    ) -> np.ndarray:
 283        """
 284        Sample from learned transition p_θ(x_{t-1} | x_t)
 285
 286        Parameters
 287        ----------
 288        xt : ndarray of shape (batch_size, n_features)
 289            Noisy data at timestep t
 290        t : int
 291            Current timestep
 292        n_samples : int
 293            Number of samples to draw per data point
 294
 295        Returns
 296        -------
 297        samples : ndarray of shape (batch_size * n_samples, n_features)
 298            Samples from learned transition
 299        """
 300        batch_size = xt.shape[0]
 301        t_array = np.full(batch_size, t)
 302
 303        # Predict noise using learned model
 304        features = self._create_features(xt, t_array)
 305        pred_noise = self.model_.predict(features).reshape(xt.shape)
 306
 307        # Compute predicted mean
 308        alpha = self.alphas[t]
 309        alpha_bar = self.alphas_cumprod[t]
 310        beta = self.betas[t]
 311
 312        coef1 = 1.0 / np.sqrt(alpha)
 313        coef2 = beta / np.sqrt(1.0 - alpha_bar)
 314        predicted_mean = coef1 * (xt - coef2 * pred_noise)
 315
 316        # Get variance
 317        if self.variance_type == "fixed_small":
 318            variance = self.posterior_variance[t]
 319        elif self.variance_type == "fixed_large":
 320            variance = beta
 321        else:  # learned - for now use fixed_small
 322            variance = self.posterior_variance[t]
 323
 324        # Sample from N(μ_θ, σ² I)
 325        if t == 0:
 326            # Deterministic at final step
 327            return np.repeat(predicted_mean, n_samples, axis=0)
 328
 329        samples = []
 330        for _ in range(n_samples):
 331            noise = self._rng.randn(*xt.shape) * np.sqrt(variance)
 332            sample = predicted_mean + noise
 333            samples.append(sample)
 334
 335        return np.vstack(samples)

 337    def _compute_kernel(
 338        self, X: np.ndarray, Y: np.ndarray, gamma: Optional[float] = None
 339    ) -> np.ndarray:
 340        """
 341        Compute kernel matrix K(X, Y) using specified kernel
 342
 343        Parameters
 344        ----------
 345        X : ndarray of shape (n, d)
 346        Y : ndarray of shape (m, d)
 347        gamma : float, optional
 348            Kernel bandwidth
 349
 350        Returns
 351        -------
 352        K : ndarray of shape (n, m)
 353            Kernel matrix
 354        """
 355        if self.mmd_kernel == "linear":
 356            return X @ Y.T
 357
 358        # Compute squared distances
 359        try:
 360            from scipy.spatial.distance import cdist
 361
 362            sq_dist = cdist(X, Y, metric="sqeuclidean")
 363        except ImportError:
 364            sq_dist = np.sum((X[:, None, :] - Y[None, :, :]) ** 2, axis=2)
 365
 366        if gamma is None:
 367            # Use median heuristic
 368            if len(X) > 100:
 369                sample_idx = self._rng.choice(len(X), 100, replace=False)
 370                sq_dist_sample = sq_dist[sample_idx][:, :100]
 371            else:
 372                sq_dist_sample = sq_dist
 373            gamma = 1.0 / np.median(sq_dist_sample[sq_dist_sample > 0])
 374
 375        if self.mmd_kernel == "rbf":
 376            return np.exp(-gamma * sq_dist)
 377        elif self.mmd_kernel == "imq":
 378            # Inverse multiquadric: (||x-y||² + c²)^(-β)
 379            c = 1.0
 380            beta = 0.5
 381            return (sq_dist + c**2) ** (-beta)
 382        else:
 383            raise ValueError(f"Unknown kernel: {self.mmd_kernel}")
 384
 385    def _compute_mmd_unbiased(
 386        self, X: np.ndarray, Y: np.ndarray, gamma: Optional[float] = None
 387    ) -> float:
 388        """
 389        Compute unbiased MMD² estimator
 390
 391        MMD²(P, Q) = E[k(u,u')] + E[k(v,v')] - 2E[k(u,v)]
 392
 393        Uses unbiased estimator that excludes diagonal terms.
 394        """
 395        n, m = len(X), len(Y)
 396
 397        # Compute kernel matrices
 398        K_XX = self._compute_kernel(X, X, gamma)
 399        K_YY = self._compute_kernel(Y, Y, gamma)
 400        K_XY = self._compute_kernel(X, Y, gamma)
 401
 402        # Unbiased estimator: exclude diagonal
 403        if n > 1:
 404            K_XX_unbiased = (K_XX.sum() - np.trace(K_XX)) / (n * (n - 1))
 405        else:
 406            K_XX_unbiased = 0
 407
 408        if m > 1:
 409            K_YY_unbiased = (K_YY.sum() - np.trace(K_YY)) / (m * (m - 1))
 410        else:
 411            K_YY_unbiased = 0
 412
 413        K_XY_mean = K_XY.mean()
 414
 415        mmd_sq = K_XX_unbiased + K_YY_unbiased - 2 * K_XY_mean
 416
 417        return max(0.0, mmd_sq)  # Ensure non-negative

 419    def _create_features(self, x: np.ndarray, t: np.ndarray) -> np.ndarray:
 420        """
 421        Create features for the regressor with multi-frequency temporal encoding
 422        """
 423        # Normalize timestep
 424        t_norm = t / self.timesteps
 425
 426        # Multi-frequency positional encoding
 427        frequencies = [1.0, 2.0, 4.0]
 428        t_encoded_parts = []
 429
 430        for freq in frequencies:
 431            t_encoded_parts.extend(
 432                [
 433                    np.sin(2 * np.pi * freq * t_norm),
 434                    np.cos(2 * np.pi * freq * t_norm),
 435                ]
 436            )
 437
 438        t_encoded_parts.append(t_norm)
 439        t_encoded = np.column_stack(t_encoded_parts)
 440
 441        # Concatenate spatial and temporal features
 442        features = np.hstack([x, t_encoded])
 443        return features

 445    def fit(
 446        self,
 447        X: np.ndarray,
 448        y: Optional[np.ndarray] = None,
 449        n_steps: int = 1000,
 450    ) -> "DiffusionModel":
 451        """
 452        Train the reverse diffusion model using specified objective
 453
 454        Parameters
 455        ----------
 456        X : ndarray of shape (n_samples, n_features)
 457            Training data (will be normalized internally)
 458        y : ignored
 459            Not used, present for API consistency
 460        n_steps : int, default=1000
 461            Number of training iterations
 462
 463        Returns
 464        -------
 465        self : DiffusionModel
 466            Fitted estimator
 467        """
 468        X = self._validate_data(X)
 469        n_samples, n_features = X.shape
 470
 471        # Store normalization params
 472        self.X_mean_ = X.mean(axis=0)
 473        self.X_std_ = X.std(axis=0) + 1e-8
 474        X_norm = (X - self.X_mean_) / self.X_std_
 475
 476        # Optional PCA
 477        if self.use_pca and n_features > 100:
 478            self.pca_ = PCA(
 479                n_components=min(self.pca_components, n_features),
 480                random_state=self.random_state,
 481            )
 482            X_norm = self.pca_.fit_transform(X_norm)
 483            n_features = X_norm.shape[1]
 484        else:
 485            self.pca_ = None
 486
 487        self.n_features_ = n_features
 488
 489        # Initialize model
 490        if self.model is None:
 491            self.model_ = Ridge(alpha=1.0, random_state=self.random_state)
 492        else:
 493            self.model_ = clone(self.model)
 494
 495        # Train based on objective
 496        if self.training_objective == "noise":
 497            self._fit_noise_prediction(X_norm, n_samples, n_steps)
 498        elif self.training_objective == "mmd":
 499            self._fit_mmd_matching(X_norm, n_samples, n_steps)
 500        elif self.training_objective == "hybrid":
 501            self._fit_hybrid(X_norm, n_samples, n_steps)
 502
 503        self.is_fitted_ = True
 504        return self

 506    def _fit_noise_prediction(
 507        self, X_norm: np.ndarray, n_samples: int, n_steps: int
 508    ) -> None:
 509        """Traditional DDPM training: predict noise with MSE loss"""
 510        n_batches = max(1, n_steps // self.batch_size)
 511
 512        X_train_batches: List[np.ndarray] = []
 513        y_train_batches: List[np.ndarray] = []
 514
 515        for batch_idx in range(n_batches):
 516            indices = self._rng.randint(n_samples, size=self.batch_size)
 517            t_batch = self._rng.randint(self.timesteps, size=self.batch_size)
 518
 519            x0_batch = X_norm[indices]
 520            noise_batch = self._rng.randn(self.batch_size, self.n_features_)
 521
 522            xt_batch, _ = self.forward_diffusion(x0_batch, t_batch, noise_batch)
 523            features = self._create_features(xt_batch, t_batch)
 524
 525            X_train_batches.append(features)
 526            y_train_batches.append(noise_batch)
 527
 528        X_train = np.vstack(X_train_batches)
 529        y_train = np.vstack(y_train_batches)
 530
 531        self.model_.fit(X_train, y_train)
 532
 533    def _fit_mmd_matching(
 534        self, X_norm: np.ndarray, n_samples: int, n_steps: int
 535    ) -> None:
 536        """
 537        MMD-based training: directly minimize distribution mismatch
 538
 539        This implements the algorithm from the theoretical document:
 540        1. Sample (x_0, t, x_t)
 541        2. Draw samples from true posterior q(x_{t-1}|x_t,x_0)
 542        3. Draw samples from learned transition p_θ(x_{t-1}|x_t)
 543        4. Compute MMD² between these sample sets
 544        5. Update θ to minimize MMD
 545        """
 546        print(
 547            f"Training with MMD objective (kernel={self.mmd_kernel}, samples={self.mmd_samples_per_step})"
 548        )
 549
 550        # Determine gamma for kernel
 551        if self.mmd_bandwidth == "auto":
 552            gamma = None  # Will be computed adaptively
 553        else:
 554            gamma = float(self.mmd_bandwidth)
 555
 556        # For sklearn models without iterative updates, we need to:
 557        # 1. Generate training data with MMD-weighted importance
 558        # 2. Fit the model on this data
 559
 560        # We'll use an iterative refinement approach:
 561        # Start with noise prediction, then refine with MMD
 562        print("Phase 1: Initial noise prediction training...")
 563        self._fit_noise_prediction(X_norm, n_samples, n_steps // 2)
 564
 565        print("Phase 2: MMD refinement...")
 566        # Generate MMD-weighted training data
 567        X_train_batches: List[np.ndarray] = []
 568        y_train_batches: List[np.ndarray] = []
 569        weights_batches: List[np.ndarray] = []
 570
 571        n_batches = max(1, (n_steps // 2) // self.batch_size)
 572
 573        for batch_idx in range(n_batches):
 574            if batch_idx % 10 == 0:
 575                print(f"  Batch {batch_idx}/{n_batches}")
 576
 577            indices = self._rng.randint(n_samples, size=self.batch_size)
 578            t_batch = self._rng.randint(1, self.timesteps, size=self.batch_size)
 579
 580            x0_batch = X_norm[indices]
 581            noise_batch = self._rng.randn(self.batch_size, self.n_features_)
 582            xt_batch, _ = self.forward_diffusion(x0_batch, t_batch, noise_batch)
 583
 584            # For each timestep, compute MMD between true and learned transitions
 585            batch_weights = []
 586            for i in range(self.batch_size):
 587                t = t_batch[i]
 588                x0_i = x0_batch[i: i + 1]
 589                xt_i = xt_batch[i: i + 1]
 590
 591                # Sample from both distributions
 592                true_samples = self._sample_true_posterior(
 593                    x0_i, xt_i, t, n_samples=self.mmd_samples_per_step
 594                )
 595                learned_samples = self._sample_learned_transition(
 596                    xt_i, t, n_samples=self.mmd_samples_per_step
 597                )
 598
 599                # Compute MMD
 600                mmd = self._compute_mmd_unbiased(
 601                    true_samples, learned_samples, gamma
 602                )
 603
 604                # Weight by MMD (higher mismatch = higher weight)
 605                weight = 1.0 + mmd
 606                batch_weights.append(weight)
 607
 608            features = self._create_features(xt_batch, t_batch)
 609            X_train_batches.append(features)
 610            y_train_batches.append(noise_batch)
 611            weights_batches.append(np.array(batch_weights))
 612
 613        # Fit with weighted samples
 614        X_train = np.vstack(X_train_batches)
 615        y_train = np.vstack(y_train_batches)
 616        weights = np.concatenate(weights_batches)
 617
 618        # Normalize weights
 619        weights = weights / weights.sum() * len(weights)
 620
 621        print("  Final weighted fit...")
 622        if (
 623            hasattr(self.model_, "fit")
 624            and "sample_weight" in self.model_.fit.__code__.co_varnames
 625        ):
 626            self.model_.fit(X_train, y_train, sample_weight=weights)
 627        else:
 628            # Fallback: repeat samples according to weights
 629            weights_int = np.round(weights).astype(int)
 630            indices_weighted = np.repeat(np.arange(len(X_train)), weights_int)
 631            self.model_.fit(
 632                X_train[indices_weighted], y_train[indices_weighted]
 633            )
 634
 635        print("MMD training complete!")

 637    def _fit_hybrid(
 638        self, X_norm: np.ndarray, n_samples: int, n_steps: int
 639    ) -> None:
 640        """Hybrid training: combine noise prediction and MMD objectives"""
 641        # Split training: 60% noise prediction, 40% MMD refinement
 642        n_noise_steps = int(0.6 * n_steps)
 643        n_mmd_steps = n_steps - n_noise_steps
 644
 645        print(
 646            f"Hybrid training: {n_noise_steps} noise steps + {n_mmd_steps} MMD steps"
 647        )
 648        self._fit_noise_prediction(X_norm, n_samples, n_noise_steps)
 649
 650        # Continue with MMD refinement
 651        self.training_objective = "mmd"  # Temporarily switch
 652        self._fit_mmd_matching(X_norm, n_samples, n_mmd_steps)
 653        self.training_objective = "hybrid"  # Restore

 655    def sample(
 656        self,
 657        n_samples: int = 1,
 658        return_trajectory: bool = False,
 659        ddim: bool = False,
 660        ddim_steps: int = 50,
 661    ) -> np.ndarray:
 662        """
 663        Generate samples via reverse diffusion
 664
 665        Parameters
 666        ----------
 667        n_samples : int, default=1
 668            Number of samples to generate
 669        return_trajectory : bool, default=False
 670            If True, return all intermediate denoising steps
 671        ddim : bool, default=False
 672            Use DDIM for faster deterministic sampling
 673        ddim_steps : int, default=50
 674            Number of DDIM steps (only used if ddim=True)
 675
 676        Returns
 677        -------
 678        samples : ndarray
 679            Generated samples (denormalized to original scale)
 680        """
 681        check_is_fitted(self, ["model_", "X_mean_", "X_std_", "n_features_"])
 682
 683        if ddim:
 684            return self._sample_ddim(n_samples, ddim_steps, return_trajectory)
 685
 686        # Standard DDPM sampling
 687        x = self._rng.randn(n_samples, self.n_features_)
 688        trajectory = [x.copy()] if return_trajectory else None
 689
 690        for t in reversed(range(self.timesteps)):
 691            t_array = np.full(n_samples, t)
 692            features = self._create_features(x, t_array)
 693
 694            pred_noise = self.model_.predict(features).reshape(x.shape)
 695
 696            alpha = self.alphas[t]
 697            alpha_bar = self.alphas_cumprod[t]
 698            beta = self.betas[t]
 699
 700            coef1 = 1.0 / np.sqrt(alpha)
 701            coef2 = beta / np.sqrt(1.0 - alpha_bar)
 702            mean = coef1 * (x - coef2 * pred_noise)
 703
 704            if t > 0:
 705                if self.variance_type == "fixed_small":
 706                    variance = self.posterior_variance[t]
 707                else:
 708                    variance = beta
 709                noise = np.sqrt(variance) * self._rng.randn(*x.shape)
 710                x = mean + noise
 711            else:
 712                x = mean
 713
 714            if return_trajectory:
 715                trajectory.append(x.copy())
 716
 717        return self._postprocess_samples(x, trajectory, return_trajectory)
 718
 719    def _sample_ddim(
 720        self, n_samples: int, steps: int, return_trajectory: bool
 721    ) -> np.ndarray:
 722        """DDIM sampling for faster generation"""
 723        skip = max(1, self.timesteps // steps)
 724        timesteps = np.arange(0, self.timesteps, skip)[::-1]
 725
 726        x = self._rng.randn(n_samples, self.n_features_)
 727        trajectory = [x.copy()] if return_trajectory else None
 728
 729        for i, t in enumerate(timesteps):
 730            t_array = np.full(n_samples, t)
 731            features = self._create_features(x, t_array)
 732            pred_noise = self.model_.predict(features).reshape(x.shape)
 733
 734            alpha_bar = self.alphas_cumprod[t]
 735            alpha_bar_prev = (
 736                self.alphas_cumprod[timesteps[i + 1]]
 737                if i < len(timesteps) - 1
 738                else 1.0
 739            )
 740
 741            pred_x0 = (x - np.sqrt(1 - alpha_bar) * pred_noise) / np.sqrt(
 742                alpha_bar
 743            )
 744            dir_xt = np.sqrt(1 - alpha_bar_prev) * pred_noise
 745            x = np.sqrt(alpha_bar_prev) * pred_x0 + dir_xt
 746
 747            if return_trajectory:
 748                trajectory.append(x.copy())
 749
 750        return self._postprocess_samples(x, trajectory, return_trajectory)

 752    def _postprocess_samples(
 753        self,
 754        x: np.ndarray,
 755        trajectory: Optional[List[np.ndarray]],
 756        return_trajectory: bool,
 757    ) -> np.ndarray:
 758        """Apply inverse transforms to generated samples"""
 759        if return_trajectory and trajectory is not None:
 760            processed_trajectory = []
 761            for step in trajectory:
 762                if self.pca_ is not None:
 763                    step = self.pca_.inverse_transform(step)
 764                step = step * self.X_std_ + self.X_mean_
 765                processed_trajectory.append(step)
 766            return np.array(processed_trajectory)
 767
 768        if self.pca_ is not None:
 769            x = self.pca_.inverse_transform(x)
 770
 771        x = x * self.X_std_ + self.X_mean_
 772        return x

 792        Parameters

 822    def _compute_mmd(
 823        self, X: np.ndarray, Y: np.ndarray, gamma: Union[str, float] = "auto"
 824    ) -> float:
 825        """Compute MMD with adaptive bandwidth (for evaluation)"""
 826        max_samples = 1000
 827        if len(X) > max_samples:
 828            indices = self._rng.choice(len(X), max_samples, replace=False)
 829            X = X[indices]
 830        if len(Y) > max_samples:
 831            indices = self._rng.choice(len(Y), max_samples, replace=False)
 832            Y = Y[indices]
 833
 834        if gamma == "auto":
